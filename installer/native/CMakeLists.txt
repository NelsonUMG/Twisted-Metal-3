cmake_minimum_required(VERSION 3.24)
project(TM3InstallerUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(TM3InstallerUI SHARED folder_picker.cpp)
target_compile_definitions(TM3InstallerUI PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(TM3InstallerUI PRIVATE /utf-8 /EHsc /W4)
target_link_libraries(TM3InstallerUI PRIVATE msi ole32 shell32 uuid)
add_executable(TM3InstallerLocationTest folder_picker_test.cpp)
target_compile_definitions(TM3InstallerLocationTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(TM3InstallerLocationTest PRIVATE /utf-8 /EHsc /W4)
target_link_libraries(TM3InstallerLocationTest PRIVATE TM3InstallerUI)
